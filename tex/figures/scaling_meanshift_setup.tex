\documentclass[tikz, border=10pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[font=\sffamily, scale=1.15]

    % K = 3 class means at the vertices of an equilateral triangle in
    % R^{K-1} = R^2, with pairwise distance 4 sigma (circumradius
    % 4/sqrt(3) = 2.309 with sigma = 1).
    \def\rad{2.309}
    \def\rng{0.8}   % P contour radius
    \def\shf{1.15}  % shift length xi, longer than the contour radius so
                    % the dotted sphere of shift directions stays visible

    \pgfmathsetmacro{\axc}{\rad*cos(90)}
    \pgfmathsetmacro{\ayc}{\rad*sin(90)}
    \pgfmathsetmacro{\bxc}{\rad*cos(210)}
    \pgfmathsetmacro{\byc}{\rad*sin(210)}
    \pgfmathsetmacro{\cxc}{\rad*cos(330)}
    \pgfmathsetmacro{\cyc}{\rad*sin(330)}

    % Shift directions v_k, one per class.
    \def\aang{40}
    \def\bang{150}
    \def\cang{290}
    \pgfmathsetmacro{\axQ}{\axc + \shf*cos(\aang)}
    \pgfmathsetmacro{\ayQ}{\ayc + \shf*sin(\aang)}
    \pgfmathsetmacro{\bxQ}{\bxc + \shf*cos(\bang)}
    \pgfmathsetmacro{\byQ}{\byc + \shf*sin(\bang)}
    \pgfmathsetmacro{\cxQ}{\cxc + \shf*cos(\cang)}
    \pgfmathsetmacro{\cyQ}{\cyc + \shf*sin(\cang)}

    % Shifted contours P_xi, drawn first so the P_0 rings sit on top.
    \foreach \x/\y in {\axQ/\ayQ, \bxQ/\byQ, \cxQ/\cyQ} {
        \fill[red!10] (\x,\y) circle (\rng);
        \draw[red!70!black, dashed, thin] (\x,\y) circle (\rng);
    }

    % Sphere of allowed shift directions around each mean.
    \foreach \x/\y in {\axc/\ayc, \bxc/\byc, \cxc/\cyc} {
        \draw[gray!65, dotted, semithick] (\x,\y) circle (\shf);
    }

    % P_0 contours and class means.
    \foreach \x/\y in {\axc/\ayc, \bxc/\byc, \cxc/\cyc} {
        \draw[blue!70!black, thick] (\x,\y) circle (\rng);
        \fill[black] (\x,\y) circle (1.5pt);
    }

    % Shift arrows xi v_k.
    \draw[-{Stealth}, gray!20!black, thick] (\axc,\ayc) -- (\axQ,\ayQ);
    \draw[-{Stealth}, gray!20!black, thick] (\bxc,\byc) -- (\bxQ,\byQ);
    \draw[-{Stealth}, gray!20!black, thick] (\cxc,\cyc) -- (\cxQ,\cyQ);

    % Mean labels, on the side opposite to each shift.
    \node[font=\footnotesize] at ($(\axc,\ayc) + (\aang+180:0.4)$) {$\mathbf{m}_0$};
    \node[font=\footnotesize] at ($(\bxc,\byc) + (\bang+180:0.4)$) {$\mathbf{m}_1$};
    \node[font=\footnotesize] at ($(\cxc,\cyc) + (\cang+180:0.4)$) {$\mathbf{m}_2$};

    % Pairwise distance, below the lowest contour.
    \pgfmathsetmacro{\dy}{\cyQ - \rng - 0.35}
    \draw[gray!60!black, {Bar[]Stealth[]}-{Stealth[]Bar[]}, thin]
        (\bxc, \dy) -- (\cxc, \dy)
        node[midway, below=1pt, font=\scriptsize, text=gray!40!black]
            {pairwise distance $4\sigma$};

    % Legend: a table centred under the drawing.
    \node[anchor=north, font=\footnotesize, inner sep=0pt] at (0, \dy-0.65) {%
        \begin{tabular}{@{}c@{\hspace{0.5em}}l@{\hspace{1.6em}}c@{\hspace{0.5em}}l@{\hspace{1.6em}}c@{\hspace{0.5em}}l@{}}
            \tikz[baseline=-0.6ex, scale=1]
                \draw[blue!70!black, thick] (0, 0) circle (4pt);
            & $\mathbb{P}_0$
            & \tikz[baseline=-0.6ex, scale=1]{
                \fill[red!10] (0, 0) circle (4pt);
                \draw[red!70!black, dashed, thin] (0, 0) circle (4pt);}
            & $\mathbb{P}_\xi$
            & \tikz[baseline=-0.6ex, scale=1]
                \draw[-{Stealth}, gray!20!black, thick] (0, 0) -- (0.45, 0);
            & shift $\xi\mathbf{v}_k$, $\|\mathbf{v}_k\| = 1$
        \end{tabular}};

\end{tikzpicture}
\end{document}
